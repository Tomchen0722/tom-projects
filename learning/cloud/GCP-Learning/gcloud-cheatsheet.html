1. 設定與認證

# 登入帳號
gcloud auth login
gcloud auth application-default login    # 應用程式憑證（SDK 用）

# 設定預設專案
gcloud config set project PROJECT_ID

# 設定預設 region 和 zone（考試常考）
gcloud config set compute/region asia-east1
gcloud config set compute/zone asia-east1-b

# 查看目前設定
gcloud config list
gcloud config get-value project

# 多組設定（不同帳號/專案切換）
gcloud config configurations create prod-config
gcloud config configurations activate prod-config
gcloud config configurations list
ACE 考試常考：gcloud config set project 和 gcloud config set compute/zone 的差異。前者設定預設專案，後者設定預設 Zone（省去每次加 --zone 旗標）。

2. Compute Engine

# 建立 VM
gcloud compute instances create my-vm \
  --machine-type=e2-medium \
  --zone=asia-east1-b \
  --image-family=debian-12 \
  --image-project=debian-cloud \
  --tags=http-server \
  --metadata-from-file startup-script=startup.sh

# 列出 VM
gcloud compute instances list
gcloud compute instances list --filter="zone:asia-east1-b"

# 啟動 / 停止 / 刪除
gcloud compute instances start my-vm --zone=asia-east1-b
gcloud compute instances stop my-vm --zone=asia-east1-b
gcloud compute instances delete my-vm --zone=asia-east1-b

# SSH 連線
gcloud compute ssh my-vm --zone=asia-east1-b

# 設定 Service Account
gcloud compute instances set-service-account my-vm \
  --service-account=sa@PROJECT.iam.gserviceaccount.com \
  --scopes=cloud-platform \
  --zone=asia-east1-b

# 防火牆規則
gcloud compute firewall-rules create allow-http \
  --direction=INGRESS --action=ALLOW \
  --rules=tcp:80 --target-tags=http-server

# 建立映像快照
gcloud compute disks snapshot my-disk \
  --snapshot-names=my-snap --zone=asia-east1-b
考試陷阱：--machine-type 改名：n1-standard-1（舊）vs e2-standard-2（新）。E2 更便宜，是現在的預設推薦。

3. GKE（Kubernetes）

# 建立 GKE Cluster
gcloud container clusters create my-cluster \
  --zone=asia-east1-b \
  --num-nodes=3 \
  --machine-type=e2-standard-2 \
  --enable-autoscaling --min-nodes=1 --max-nodes=5

# Autopilot 模式（推薦，無需管 Node）
gcloud container clusters create-auto my-autopilot \
  --region=asia-east1

# 取得 kubectl 認證設定
gcloud container clusters get-credentials my-cluster --zone=asia-east1-b

# 升級 Cluster 版本
gcloud container clusters upgrade my-cluster --master --zone=asia-east1-b

# 列出所有 Cluster
gcloud container clusters list

# 刪除 Cluster
gcloud container clusters delete my-cluster --zone=asia-east1-b

# kubectl 常用（取得認證後）
kubectl get pods --all-namespaces
kubectl get services
kubectl apply -f deployment.yaml
kubectl scale deployment my-app --replicas=5
ACE 重點：get-credentials 是讓本地端的 kubectl 連到 GKE Cluster 的關鍵指令。沒有這一步，kubectl 指令都找不到 Cluster。

4. Cloud Storage（gsutil / gcloud storage）

# 建立 Bucket（新語法用 gcloud storage）
gcloud storage buckets create gs://my-bucket \
  --location=asia-east1 \
  --default-storage-class=STANDARD

# 上傳 / 下載 / 列出（gsutil 仍常用）
gsutil cp local-file.txt gs://my-bucket/
gsutil cp gs://my-bucket/file.txt ./
gsutil ls gs://my-bucket/
gsutil ls -la gs://my-bucket/          # 顯示版本（版本控制開啟時）
gsutil rm gs://my-bucket/file.txt

# 遞迴複製（同步整個目錄）
gsutil -m rsync -r ./local-dir gs://my-bucket/dir/

# 設定 ACL（現在推薦用 IAM 不用 ACL）
gsutil iam ch allUsers:objectViewer gs://my-bucket  # 公開讀取

# 生命週期設定
gsutil lifecycle set lifecycle.json gs://my-bucket

# 查看 Bucket 設定
gsutil ls -L -b gs://my-bucket
考試陷阱：gsutil cp 預設不遞迴。複製整個資料夾要加 -r 旗標：gsutil cp -r ./dir gs://bucket/

5. Cloud SQL

# 建立 Cloud SQL Instance（MySQL）
gcloud sql instances create my-db \
  --database-version=MYSQL_8_0 \
  --tier=db-n1-standard-2 \
  --region=asia-east1 \
  --availability-type=REGIONAL   # REGIONAL = 高可用（自動備援）

# 建立資料庫和使用者
gcloud sql databases create mydb --instance=my-db
gcloud sql users create app-user --instance=my-db --password=secret

# 連線到 Cloud SQL（使用 Cloud SQL Auth Proxy）
./cloud-sql-proxy my-project:asia-east1:my-db &
mysql -u app-user -p -h 127.0.0.1

# 或直接用 gcloud（Cloud Shell 環境）
gcloud sql connect my-db --user=app-user

# 備份
gcloud sql backups create --instance=my-db
gcloud sql backups list --instance=my-db

# 列出所有 Instance
gcloud sql instances list

6. BigQuery

# 建立 Dataset
bq mk --location=asia-east1 my_dataset

# 執行查詢（計費按掃描量）
bq query --use_legacy_sql=false --dry_run 'SELECT * FROM dataset.table'
# --dry_run 不實際執行，只顯示預估掃描量（考試重點！）

# 載入資料（從 GCS）
bq load --source_format=CSV --autodetect my_dataset.my_table gs://bucket/data.csv

# 匯出資料
bq extract my_dataset.my_table gs://bucket/export-*.csv

# 列出 Dataset 和 Table
bq ls
bq ls my_dataset
bq show my_dataset.my_table    # 顯示 Schema

# 刪除 Table
bq rm -t my_dataset.my_table
考試重點：bq query --dry_run 估算費用（BigQuery 按掃描量計費）。加 --maximum_bytes_billed 設上限，超過就取消查詢，防止意外高費用。

7. IAM 與 Service Account

# 查看 IAM 政策
gcloud projects get-iam-policy PROJECT_ID

# 授予角色
gcloud projects add-iam-policy-binding PROJECT_ID \
  --member="user:alice@example.com" \
  --role="roles/storage.objectViewer"

# 移除角色
gcloud projects remove-iam-policy-binding PROJECT_ID \
  --member="user:alice@example.com" \
  --role="roles/storage.objectViewer"

# Service Account 操作
gcloud iam service-accounts create my-sa \
  --display-name="My Service Account"

# 下載 SA 金鑰（JSON）——盡量避免，改用 Workload Identity
gcloud iam service-accounts keys create key.json \
  --iam-account=my-sa@PROJECT.iam.gserviceaccount.com

# 列出 Service Account
gcloud iam service-accounts list

# 模擬執行（檢查是否有某個權限）
gcloud iam list-testable-permissions //cloudresourcemanager.googleapis.com/projects/PROJECT_ID
最佳實踐：盡量用 Workload Identity（GKE）或 VM Service Account 附加，不要下載 JSON Key。JSON Key 洩漏等於服務帳號被攻陷，很難追蹤。

8. 網路

# 建立自訂 VPC
gcloud compute networks create my-vpc --subnet-mode=custom

# 建立子網路
gcloud compute networks subnets create my-subnet \
  --network=my-vpc --region=asia-east1 --range=10.0.1.0/24

# 建立 Cloud NAT（讓私有 VM 可以對外連網）
gcloud compute routers create my-router --network=my-vpc --region=asia-east1
gcloud compute routers nats create my-nat \
  --router=my-router --region=asia-east1 \
  --nat-all-subnet-ip-ranges --auto-allocate-nat-external-ips

# Cloud DNS（私有 Zone）
gcloud dns managed-zones create my-zone \
  --dns-name=internal.example.com \
  --visibility=private --networks=my-vpc

# VPC Peering
gcloud compute networks peerings create peer-a-to-b \
  --network=vpc-a --peer-network=vpc-b

9. Cloud Run / Functions

# 部署 Cloud Run 服務（從映像）
gcloud run deploy my-service \
  --image=asia-east1-docker.pkg.dev/PROJECT/repo/image:latest \
  --region=asia-east1 \
  --allow-unauthenticated \
  --min-instances=1 \
  --max-instances=10 \
  --concurrency=80 \
  --memory=512Mi \
  --cpu=1

# 列出服務
gcloud run services list
gcloud run services describe my-service --region=asia-east1

# 更新流量分配（藍綠部署）
gcloud run services update-traffic my-service \
  --to-revisions=my-service-v2=80,my-service-v1=20 \
  --region=asia-east1

# Cloud Functions（2nd Gen）
gcloud functions deploy my-func \
  --gen2 --runtime=python311 \
  --region=asia-east1 \
  --entry-point=main \
  --trigger-http --allow-unauthenticated

# 查看函式日誌
gcloud functions logs read my-func --region=asia-east1

10. 考試常見情境指令

情境指令
找出所有 VM 的外部 IPgcloud compute instances list --format="table(name,networkInterfaces[0].accessConfigs[0].natIP)"
設定 VM 的啟動腳本gcloud compute instances create ... --metadata=startup-script='#!/bin/bash\napt-get install -y nginx'
確認有效的 gcloud 帳號gcloud auth list
查看 API 是否啟用gcloud services list --enabled
啟用某個 APIgcloud services enable compute.googleapis.com
把 GCS 物件設為公開gsutil iam ch allUsers:objectViewer gs://BUCKET
建立排程的 Cloud Run Jobgcloud scheduler jobs create http ... --schedule="0 9 * * 1"
查看帳單帳號gcloud billing accounts list
把帳單帳號連結到專案gcloud billing projects link PROJECT_ID --billing-account=ACCOUNT_ID
查看 Quota 使用量gcloud compute project-info describe --project=PROJECT_ID
考試技巧：ACE 考題常用 gcloud [服務] [資源] [動作] 格式。記住資源名稱（如 instances 不是 vms、clusters 不是 kubernetes）。動作：create / list / describe / delete / update。
旗標記憶法--zone：單一機房（如 asia-east1-b）
--region：整個地區（如 asia-east1）
Regional 資源用 --region，Zonal 資源用 --zone
--format 輸出格式--format=json：腳本處理
--format=csv：試算表匯入
--format="table(name,zone)"：自訂欄位
--filter 篩選--filter="status=RUNNING"
--filter="zone:asia-east1-b"
--filter="labels.env=prod"
